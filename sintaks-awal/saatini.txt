-- ============================================================
-- Program: Bilangan Prima
-- Deskripsi: Mengecek dan menampilkan bilangan prima
-- ============================================================

fungsi adalah_prima(n: bilangan) -> logika {
    jika n <= 1 maka {
        kembalikan salah
    }

    jika n <= 3 maka {
        kembalikan benar
    }

    jika n % 2 == 0 atau n % 3 == 0 maka {
        kembalikan salah
    }

    variabel i: bilangan = 5
    selama i * i <= n {
        jika n % i == 0 atau n % (i + 2) == 0 maka {
            kembalikan salah
        }
        i = i + 6
    }

    kembalikan benar
}

fungsi utama() -> bilangan {
    tampilkan "=== BILANGAN PRIMA ===\n"

    -- Menampilkan bilangan prima 1-100
    tampilkan "Bilangan prima antara 1-100:\n"

    variabel hitung: bilangan = 0
    untuk i dari 1 sampai 100 {
        jika adalah_prima(i) maka {
            jika hitung > 0 maka {
                tampilkan ", "
            }
            tampilkan i
            hitung = hitung + 1
        }
    }

    tampilkan "\nTotal: ", hitung, " bilangan prima\n"

    kembalikan 0
}

--- ./scope_dan_shadowing.rhc ----------
-- ============================================================
-- Tes: Scope dan Shadowing Variabel
-- ============================================================

variabel global_var: bilangan = 100

fungsi utama() -> bilangan {
    tampilkan "=== TEST SCOPE DAN SHADOWING ===\n"

    tampilkan "Global var: ", global_var, "\n"

    -- Variabel lokal
    variabel lokal: bilangan = 50
    tampilkan "Variabel lokal di main: ", lokal, "\n"

    -- Blok scope
    {
        variabel dalam_blok: bilangan = 25
        tampilkan "  Dalam blok - dalam_blok: ", dalam_blok, "\n"
        tampilkan "  Dalam blok - lokal (dari main): ", lokal, "\n"
        tampilkan "  Dalam blok - global_var: ", global_var, "\n"

        -- Shadowing
        variabel lokal: bilangan = 999
        tampilkan "  Dalam blok - lokal (shadow): ", lokal, "\n"
    }

    -- lokal kembali ke nilai semula
    tampilkan "Kembali ke main - lokal: ", lokal, "\n"

    -- Loop scope
    tampilkan "Loop scope:\n"

    untuk i dari 1 sampai 3 {
        variabel loop_var: bilangan = i * 10
        tampilkan "  i = ", i, ", loop_var = ", loop_var, "\n"
    }

    -- Nested blocks
    tampilkan "Nested blocks:\n"

    {
        variabel a: bilangan = 1
        tampilkan "  Block 1: a = ", a, "\n"

        {
            variabel b: bilangan = 2
            variabel a: bilangan = 10  -- shadow
            tampilkan "    Block 2: a = ", a, ", b = ", b, "\n"
        }

        tampilkan "  Block 1: a = ", a, "\n"
    }

    -- Fungsi scope (simbol global vs lokal)
    tampilkan "Global var tetap: ", global_var, "\n"

    kembalikan 0
}

-- ============================================================
-- Program: Array (Larik)
-- Deskripsi: Demonstrasi penggunaan larik
-- ============================================================

fungsi utama() -> bilangan {
    tampilkan "=== DEMONSTRASI LARIK ===\n"

    -- Deklarasi larik
    variabel angka: larik[5] dari bilangan

    -- Inisialisasi larik
    angka[0] = 10
    angka[1] = 20
    angka[2] = 30
    angka[3] = 40
    angka[4] = 50

    -- Menampilkan isi larik
    tampilkan "Isi larik:\n"
    untuk i dari 0 sampai 4 {
        tampilkan "  angka[", i, "] = ", angka[i], "\n"
    }

    -- Menghitung jumlah dan rata-rata
    variabel total: bilangan = 0
    untuk i dari 0 sampai 4 {
        total = total + angka[i]
    }

    variabel rata: bilangan = total / 5

    tampilkan "Total: ", total, "\n"
    tampilkan "Rata-rata: ", rata, "\n"

    -- Mencari nilai maksimum
    variabel maks: bilangan = angka[0]
    untuk i dari 1 sampai 4 {
        jika angka[i] > maks maka {
            maks = angka[i]
        }
    }
    tampilkan "Nilai maksimum: ", maks, "\n"

    kembalikan 0
}

-- tes Generik (Generics)

-- Fungsi generik dengan constraint
generik T fungsi maks(a: T, b: T) -> T
dimana T: Terbandingkan
{
    kembalikan a > b ? a : b
}

-- Struktur generik
generik T struktur Pasangan {
    kiri: T,
    kanan: T
}

-- Cacahan generik
generik T cacahan Opsi {
    Ada(T),
    Kosong
}

-- Prosedur utama
prosedur utama() {
    -- Pemanggilan eksplisit
    variabel m: bilangan = maks(dengan T = bilangan, 3, 4)
    tampilkan m, "\n"

    -- Struktur generik
    variabel p: Pasangan(dengan T = teks)
    p.kiri = "halo"
    p.kanan = "dunia"
    tampilkan p.kiri, " ", p.kanan, "\n"

    -- Cacahan generik
    variabel o: Opsi(dengan T = bilangan) = Ada(42)

    -- Lambda generik
    variabel f = lambda(generik T: Terbandingkan)(x: T) {
        kembalikan x
    }
    tampilkan f(5), "\n"
}

--- ./sum_type.rhc ----------
-- Tes Sum Type

-- Opsi harus didefinisikan lebih dulu
generik T cacahan_sum Opsi
dimana T: Terbandingkan
{
    kasus ADA(nilai: T),
    kasus KOSONG()
}

-- Hasil
generik T, E cacahan_sum Hasil {
    kasus SUKSES(nilai: T),
    kasus GAGAL(pesan: E)
}

-- Pohon rekursif lewat Opsi
cacahan_sum POHON {
    kasus DAUN(nilai: bilangan),
    kasus CABANG(anak: Opsi(POHON))
}

-- Struktur berisi sum type
struktur Pembungkus {
    data: Hasil(dengan T = bilangan, E = teks),
    panjang: bilangan
}

-- Pemakaian di dalam cacahan
cacahan Daftar_Tipe {
    OPSI_BILANGAN = cacahan_sum(Opsi(dengan T = bilangan))
}

-- Program utama
prosedur utama() {
    variabel x: Opsi(dengan T = bilangan) = ADA(nilai = 42)
    variabel y: Opsi(dengan T = bilangan) = KOSONG()

    variabel hasil: Hasil(dengan T = bilangan, E = teks) =
        SUKSES(nilai = 100)

    pilih hasil {
        kasus SUKSES(nilai):
            tampilkan "Sukses: ", nilai, "\n"
        kasus GAGAL(pesan):
            tampilkan "Gagal: ", pesan, "\n"
    }

    -- Pengambilan lewat cacahan
    variabel z: Daftar_Tipe.OPSI_BILANGAN = ADA(nilai = 7)
    pilih z {
        kasus ADA(nilai):
            tampilkan "Daftar_Tipe.ADA: ", nilai, "\n"
        kasus KOSONG():
            tampilkan "Daftar_Tipe.KOSONG\n"
    }

    -- Pembacaan otomatis
    tampilkan "ambil_ada(x) = ", ambil_ada(x), "\n"

    -- Struktur berisi sum type
    variabel p: Pembungkus
    p.data = SUKSES(nilai = 200)
    p.panjang = 3

    pilih p.data {
        kasus SUKSES(nilai):
            tampilkan "Pembungkus.SUKSES: ", nilai, "\n"
        kasus GAGAL(pesan):
            tampilkan "Pembungkus.GAGAL: ", pesan, "\n"
    }
}

-- Pengujian fitur 'lambda(...) { ... }'

-- Fungsi tingkat-tinggi: menerima lambda sebagai argumen (callback)
fungsi terapkan(f: tunjuk ke fungsi, x: bilangan, y: bilangan) -> bilangan {
    kembalikan f(x, y)
}

prosedur utama() {
    -- Lambda disimpan ke variabel
    tampilkan "Lambda disimpan dalam variabel:\n"
    variabel jumlah = lambda(a: bilangan, b: bilangan) { kembalikan a + b }
    tampilkan jumlah(3, 4), "\n"

    -- Lambda dengan ternary di dalam badannya
    tampilkan "\nLambda dengan ternary:\n"
    variabel maksimum = lambda(a: bilangan, b: bilangan) { kembalikan a > b ? a : b }
    tampilkan maksimum(10, 20), "\n"
    tampilkan maksimum(99, 3), "\n"

    -- Lambda dioper LANGSUNG sebagai argumen ke fungsi lain (callback)
    tampilkan "\nLambda sebagai argumen ke fungsi lain (callack):\n"
    tampilkan terapkan(lambda(a: bilangan, b: bilangan) { kembalikan a * b }, 6, 7), "\n"
    tampilkan terapkan(lambda(a: bilangan, b: bilangan) { kembalikan a < b ? a : b }, 15, 8), "\n"

    -- Lambda dengan operasi satu baris lain: perbandingan (logika)
    tampilkan "\nLambda dengan operasi satu baris:\n"
    variabel sama = lambda(a: bilangan, b: bilangan) { kembalikan a == b }
    jika sama(5, 5) maka {
        tampilkan 1, "\n"
    } jika bukan semua diatas {
        tampilkan 0, "\n"
    }

    -- Lambda tanpa parameter
    tampilkan "\nLambda tanpa parameter:\n"
    variabel konstanta = lambda() { kembalikan 777 }
    tampilkan konstanta(), "\n"
}

    }

    -- LANGKAH 3: Menghapus File
    tampilkan "\n[3] Menghapus file \'", contoh, "\'...\n"

    variabel sukses_hapus: logika

    hapus_file(contoh) -> sukses_hapus

    jika sukses_hapus maka {
        tampilkan "  ✓ File berhasil dihapus!\n"
    } jika bukan semua diatas {
        tampilkan "  ✗ Gagal menghapus file!\n"
        kembalikan 1
    }

    -- LANGKAH 4: Verifikasi File Telah Dihapus (Opsional)
    tampilkan "\n[4] Verifikasi file telah dihapus (Opsional)...\n"

    variabel isi_setelah_hapus: teks
    variabel sukses_setelah_hapus: logika

    baca_file(contoh, isi_setelah_hapus) -> sukses_setelah_hapus

    jika sukses_setelah_hapus == 0 maka {
        tampilkan "  ✓ File tidak dapat dibaca (berhasil dihapus)\n"
    } jika bukan semua diatas {
        tampilkan "  ⚠ File masih ada (penghapusan mungkin gagal)\n"
    }

    tampilkan "║              DEMONSTRASI SELESAI!                        ║\n"
    tampilkan "╚══════════════════════════════════════════════════════════╝\n"

prosedur contoh_alokasi() {
    variabel ptr: tunjuk ke bilangan = alokasi(bilangan)
    isi(ptr) = 100
    goto clean_up

    clean_up: {
        tampilkan isi(ptr)
        bebaskan(ptr)
    }
}

-- Contoh 2: early return + label pembersihan (kembalikan diikuti label)
fungsi cari(x: bilangan) -> bilangan {
    jika x < 0 maka {
        goto gagal
    }
    kembalikan x * 2

    gagal: {
        kembalikan -1
    }
}

-- Contoh 3: lompatan mundur (loop manual via goto)
prosedur hitung_mundur() {
    variabel i: bilangan = 5
    ulang: {
        tampilkan i
        i = i - 1
        jika i > 0 maka {
            goto ulang
        }
    }
}

prosedur utama() {
    contoh_alokasi()
    tampilkan cari(21), "\n"
    tampilkan cari(-5), "\n"
    hitung_mundur()
    tampilkan "\n"
}

-- ============================================================
-- Tes: Fungsi Rekursif
-- ============================================================

-- ============================================================
-- Mutual Recursion: is_even dan is_odd
-- ============================================================
deklarasi fungsi is_odd(n: bilangan) -> logika;
fungsi is_even(n: bilangan) -> logika {
    jika n == 0 maka {
        kembalikan benar
    }
    kembalikan is_odd(n - 1)
}

fungsi is_odd(n: bilangan) -> logika {
    jika n == 0 maka {
        kembalikan salah
    }
    kembalikan is_even(n - 1)
}

-- ============================================================
-- Mutual Recursion: m dan f (Hofstadter Female/Male)
-- ============================================================
deklarasi fungsi male(n: bilangan) -> bilangan;
fungsi female(n: bilangan) -> bilangan {
    jika n == 0 maka {
        kembalikan 1
    }
    kembalikan n - male(female(n - 1))
}

fungsi male(n: bilangan) -> bilangan {
    jika n == 0 maka {
        kembalikan 0
    }
    kembalikan n - female(male(n - 1))
}

-- ============================================================
-- SINGLE RECURSION (Non-Tail)
-- ============================================================
fungsi faktorial(n: bilangan) -> bilangan {
    jika n <= 1 maka {
        kembalikan 1
    }
    kembalikan n * faktorial(n - 1)
}

fungsi fibonacci(n: bilangan) -> bilangan {
    jika n <= 1 maka {
        kembalikan n
    }
    kembalikan fibonacci(n - 1) + fibonacci(n - 2)
}

-- ============================================================
-- TAIL RECURSION FUNCTIONS
-- ============================================================
-- Tail Recursion: Faktorial dengan akumulator
fungsi faktorial_tail(n: bilangan, akumulator: bilangan) -> bilangan {
    jika n <= 1 maka {
        kembalikan akumulator
    }
    kembalikan faktorial_tail(n - 1, n * akumulator)
}

-- Wrapper untuk faktorial tail recursion
fungsi faktorial_tr(n: bilangan) -> bilangan {
    kembalikan faktorial_tail(n, 1)
}

-- Tail Recursion: Fibonacci dengan dua akumulator
fungsi fibonacci_tail(n: bilangan, a: bilangan, b: bilangan) -> bilangan {
    jika n == 0 maka {
        kembalikan a
    }
    jika n == 1 maka {
        kembalikan b
    }
    kembalikan fibonacci_tail(n - 1, b, a + b)
}

-- Wrapper untuk fibonacci tail recursion
fungsi fibonacci_tr(n: bilangan) -> bilangan {
    kembalikan fibonacci_tail(n, 0, 1)
}

-- Tail Recursion: Pangkat dengan akumulator
fungsi pangkat_tail(basis: bilangan, eksp: bilangan, akumulator: bilangan) -> bilangan {
    jika eksp == 0 maka {
        kembalikan akumulator
    }
    kembalikan pangkat_tail(basis, eksp - 1, basis * akumulator)
}

-- Wrapper untuk pangkat tail recursion
fungsi pangkat_tr(basis: bilangan, eksp: bilangan) -> bilangan {
    kembalikan pangkat_tail(basis, eksp, 1)
}

-- Tail Recursion: Jumlah digit dengan akumulator
fungsi jumlah_digit_tail(n: bilangan, akumulator: bilangan) -> bilangan {
    jika n == 0 maka {
        kembalikan akumulator
    }
    kembalikan jumlah_digit_tail(n / 10, akumulator + (n % 10))
}

-- Wrapper untuk jumlah digit tail recursion
fungsi jumlah_digit_tr(n: bilangan) -> bilangan {
    kembalikan jumlah_digit_tail(n, 0)
}

-- Tail Recursion: GCD dengan Euclidean algorithm (sudah tail recursive)
fungsi gcd_tail(a: bilangan, b: bilangan) -> bilangan {
    jika b == 0 maka {
        kembalikan a
    }
    kembalikan gcd_tail(b, a % b)
}

-- Tail Recursion: Sum dari 1 ke n dengan akumulator
fungsi sum_tail(n: bilangan, akumulator: bilangan) -> bilangan {
    jika n == 0 maka {
        kembalikan akumulator
    }
    kembalikan sum_tail(n - 1, akumulator + n)
}

-- Wrapper untuk sum tail recursion
fungsi sum_tr(n: bilangan) -> bilangan {
    kembalikan sum_tail(n, 0)
}

-- Tail Recursion: Reverse number (membalik digit)
fungsi reverse_tail(n: bilangan, akumulator: bilangan) -> bilangan {
    jika n == 0 maka {
        kembalikan akumulator
    }
    kembalikan reverse_tail(n / 10, akumulator * 10 + (n % 10))
}

-- Wrapper untuk reverse tail recursion
fungsi reverse_tr(n: bilangan) -> bilangan {
    kembalikan reverse_tail(n, 0)
}

-- Tail Recursion: Cek palindrome number
fungsi is_palindrome(n: bilangan) -> logika {
    kembalikan n == reverse_tr(n)
}

-- Tail Recursion: Binary representation (jumlah digit biner)
fungsi count_binary_digits_tail(n: bilangan, akumulator: bilangan) -> bilangan {
    jika n == 0 maka {
        kembalikan akumulator
    }
    kembalikan count_binary_digits_tail(n / 2, akumulator + 1)
}

-- Wrapper
fungsi count_binary_digits(n: bilangan) -> bilangan {
    jika n == 0 maka {
        kembalikan 1
    }
    kembalikan count_binary_digits_tail(n, 0)
}

        tampilkan i, "\t", female(i), "\t", male(i), "\n"
    }

    kembalikan 0

-- Implementasi fungsi tambah
fungsi tambah(a: bilangan, b: bilangan) -> bilangan {
    kembalikan a + b
}

-- Implementasi fungsi kali
fungsi kali(a: bilangan, b: bilangan) -> bilangan {
    kembalikan a * b
}

prosedur tampilkan_pesan(pesan: teks) {
    tampilkan "Pesan: ", pesan, "\n"
}

-- Fungsi yang menggunakan fungsi lain (didefinisikan setelah)
fungsi hitung(a: bilangan, b: bilangan) -> bilangan {
    variabel hasil: bilangan

    hasil = tambah(a, b)
    hasil = kali(hasil, 2)

    kembalikan hasil
}

    tampilkan_pesan("Testing prototipe")

    variabel x: bilangan = 10
    variabel y: bilangan = 5

    tampilkan "tambah(", x, ", ", y, ") = ", tambah(x, y), "\n"
    tampilkan "kali(", x, ", ", y, ") = ", kali(x, y), "\n"
    tampilkan "hitung(", x, ", ", y, ") = ", hitung(x, y), "\n"

    -- AND bitwise
    variabel hasil_dan: bilangan = a & b
    tampilkan "a & b  = ", hasil_dan, " (0x05, biner: 00000101)\n"

    -- OR bitwise
    variabel hasil_atau: bilangan = a | b

    -- XOR bitwise

    tampilkan "a ^ b  = ", hasil_xor, " (0x5A, biner: 01011010)\n"

    -- NOT bitwise
    variabel hasil_not: bilangan = ~a
    tampilkan "~a     = ", hasil_not, " (komplemen dari 0x0F)\n"

    -- Geser kiri
    variabel geser_kiri: bilangan = a << 2

    -- Geser kanan
    variabel geser_kanan: bilangan = b >> 2

    tampilkan "x awal = ", x, "\n"

    x &= 0x05
    tampilkan "x &= 0x05 -> ", x, "\n"

    x |= 0xA0
    tampilkan "x |= 0xA0 -> ", x, "\n"

    x ^= 0xFF
    tampilkan "x ^= 0xFF -> ", x, "\n"

    x <<= 1
    tampilkan "x <<= 1 -> ", x, "\n"

    x >>= 2
    tampilkan "x >>= 2 -> ", x, "\n"

    -- Flag manipulation
    tampilkan "\n=== MANIPULASI FLAG ===\n"

    flags = flags | (1 << 2)

pembangkit fungsi hitung_naik(batas: bilangan) -> Pembangkit(dengan T = bilangan, R = hampa) {
    variabel i: bilangan = 0
    selama i < batas {
        hasil(i)
        i = i + 1
    }
}

-- Pembangkit banyak nilai
pembangkit fungsi pasangan() -> Pembangkit(dengan T = bilangan, R = hampa) {
    hasil(1, 2, 3)
    hasil(4, 5, 6)
}

-- Pembangkit delegasi
pembangkit fungsi gabung(a: Pembangkit(dengan T = bilangan, R = hampa),
                         b: Pembangkit(dengan T = bilangan, R = hampa))
    -> Pembangkit(dengan T = bilangan, R = hampa)
{
    untuk setiap x dalam a {
        hasil(x)
    }
    untuk setiap x dalam b {
        hasil(x)
    }
}

-- Pembangkit rekursif
pembangkit fungsi turun(n: bilangan) -> Pembangkit(dengan T = bilangan, R = hampa) {
    jika n <= 0 maka {
        hasil(0)
    } jika bukan semua diatas {
        hasil(n)
        hasil dari turun(n - 1)
    }
}

-- Pembangkit dengan pilih
pembangkit fungsi kategori(x: bilangan) -> Pembangkit(dengan T = bilangan, R = hampa) {
    pilih x {
        kasus 1: {
            hasil(100)
        }
        kasus 2: {
            hasil(200)
        }
        kasus 3: {
            hasil(300)
        }
    }
}

-- 1. Korutin dasar dengan jeda
coroutine fungsi pencetak(batas: bilangan) -> Pembangkit(dengan T = bilangan, R = hampa) {
    variabel i: bilangan = 0
    selama i < batas {
        hasil(i)
        jeda
        i = i + 1
    }
}

-- 2. Korutin tanpa jeda, mirip pembangkit
coroutine fungsi deret_ganjil(batas: bilangan) -> Pembangkit(dengan T = bilangan, R = hampa) {
    variabel i: bilangan = 1
    selama i <= batas {
        hasil(i)
        i = i + 2
    }
}

-- 3. Korutin dengan `jeda sampai kondisi`
coroutine fungsi tunggu_sampai(ambang: bilangan) -> Pembangkit(dengan T = bilangan, R = hampa) {
    variabel x: bilangan = 0
    selama benar {
        x = x + 1
        jeda sampai x >= ambang
        hasil(x)
        jika x >= ambang * 2 maka {
            berhenti
        }
    }
}

-- 4. Korutin dengan pilih
coroutine fungsi kategori(n: bilangan) -> Pembangkit(dengan T = teks, R = hampa) {

            jeda
            hasil("satu lagi")
        }
        kasus 2: {
            hasil("dua")
        }
        kasus 3: {
            hasil("tiga")
            jeda
            hasil("tiga lagi")
        }
    }
}

-- 5. Korutin dengan loop `untuk setiap`
coroutine fungsi salin_dari(sumber: Pembangkit(dengan T = bilangan, R = hampa))
    -> Pembangkit(dengan T = bilangan, R = hampa)
{

        hasil(x * 10)
        jeda
    }
}

-- 6. Korutin rekursif
coroutine fungsi turun(n: bilangan) -> Pembangkit(dengan T = bilangan, R = hampa) {
    jika n <= 0 maka {
        hasil(0)
    } jika bukan semua diatas {
        hasil(n)
        jeda
        hasil dari turun(n - 1)
    }
}

-- 7. Korutin dengan ulangi
coroutine fungsi ulang_tiga() -> Pembangkit(dengan T = bilangan, R = hampa) {
    variabel i: bilangan = 0
    ulangi {
        hasil(i)
        jeda
        i = i + 1
    } selama i < 3
}

-- ============================================================
-- Tes Mutex
-- ============================================================

-- Data bersama yang dilindungi mutex
tugas_global variabel saldo: bilangan = 1000
tugas_global variabel mutex_saldo: Mutex = mutex()

-- Riwayat transaksi
tugas_global variabel riwayat: Daftar(dengan T = teks, jenis = tunggal) = daftar()
tugas_global variabel mutex_riwayat: Mutex = mutex()

-- Fungsi: tarik saldo
fungsi tarik(jumlah: bilangan) -> Opsi(dengan T = bilangan) {
    mutex_kunci(mutex_saldo)

    jamin saat keluar {
        mutex_buka(mutex_saldo)
    }

    jika jumlah > saldo maka {
        mutex_kunci(mutex_riwayat)
        daftar_tambah(riwayat, "Gagal tarik: saldo tidak cukup")
        mutex_buka(mutex_riwayat)
        kembalikan KOSONG()
    }

    saldo = saldo - jumlah

    mutex_kunci(mutex_riwayat)
    daftar_tambah(riwayat, "Tarik berhasil")
    mutex_buka(mutex_riwayat)

    kembalikan ADA(nilai = saldo)
}

-- Fungsi: setor saldo
fungsi setor(jumlah: bilangan) -> bilangan {
    mutex_kunci(mutex_saldo)

    jamin saat keluar {
        mutex_buka(mutex_saldo)
    }

    saldo = saldo + jumlah

    mutex_kunci(mutex_riwayat)
    daftar_tambah(riwayat, "Setor berhasil")
    mutex_buka(mutex_riwayat)

    kembalikan saldo
}

-- Fungsi: tampilkan saldo
fungsi lihat_saldo() -> bilangan {
    mutex_kunci(mutex_saldo)

    jamin saat keluar {
        mutex_buka(mutex_saldo)
    }

    kembalikan saldo
}

-- Program utama
prosedur utama() {
    tampilkan "=== TEST MUTEX ===\n"

    -- Saldo awal
    tampilkan "Saldo awal: ", lihat_saldo(), "\n"

    variabel s1: Pekerja(dengan T = bilangan) = buat_pekerja(setor, 500)

    pilih pekerja_tunggu(s1) {
        kasus ADA(nilai):
            tampilkan "Setor 500 -> saldo: ", nilai, "\n"

            tampilkan "Setor gagal\n"
    }

    pilih pekerja_tunggu(s2) {
        kasus ADA(nilai):
            tampilkan "Setor 300 -> saldo: ", nilai, "\n"
        kasus KOSONG():
            tampilkan "Setor gagal\n"

    -- Tarik

        kasus ADA(nilai):
            tampilkan "Tarik 200 -> saldo: ", nilai, "\n"

            tampilkan "Tarik gagal\n"

    -- Beberapa pekerja sekaligus

            tampilkan "Dari pointer: ", nilai, "\n"
        kasus KOSONG():
            tampilkan "Kosong\n"
    }

    saluran_tutup(isi(ptr_ch))
    bebaskan(ptr_ch)

    tampilkan "\n=== SELESAI ===\n"

-- 1. Fungsi dengan parameter big integer
fungsi jumlah_btb(a: bilangan_tanpa_batas, b: bilangan_tanpa_batas) -> bilangan_tanpa_batas {
    kembalikan a + b
}

-- 2. Fungsi mengembalikan big integer
fungsi faktorial_btb(n: bilangan) -> bilangan_tanpa_batas {
    variabel hasil: bilangan_tanpa_batas("1")
    variabel i: bilangan = 2
    selama i <= n {
        hasil = hasil * btb(ke_teks(i))
        i = i + 1
    }
    kembalikan hasil
}

-- 3. Fungsi fibonacci dengan big integer
fungsi fibonacci_btb(n: bilangan) -> bilangan_tanpa_batas {
    variabel a: bilangan_tanpa_batas("0")
    variabel b: bilangan_tanpa_batas("1")
    variabel i: bilangan = 0
    selama i < n {
        variabel c = a + b
        a = b
        b = c
        i = i + 1
    }
    kembalikan a
}

-- 4. Fungsi untuk menampilkan big integer
prosedur tampilkan_btb(label: teks, x: bilangan_tanpa_batas) {
    tampilkan label, " = ", ke_teks(x), "\n"
}
